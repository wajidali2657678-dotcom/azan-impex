function getFilteredProducts(sortOverride) {
let products = [...PRODUCTS];
// Category
if (currentCat !== 'all') products = products.filter(p => p.category === currentCat);
// Search
if (searchQuery) {
const q = searchQuery.toLowerCase();
products = products.filter(p => p.name.toLowerCase().includes(q) || p.sku.toLowerCase().includes(q) || p.fabric.toLowerCase().includes(q) || p.category.toLowerCase().includes(q) || p.format.toLowerCase().includes(q));
}
// Filters
if (activeFilters.category.length) products = products.filter(p => activeFilters.category.includes(p.category));
if (activeFilters.fabric.length) products = products.filter(p => activeFilters.fabric.includes(p.fabric));
if (activeFilters.gsm.length) products = products.filter(p => activeFilters.gsm.includes(p.gsm));
if (activeFilters.status.length) products = products.filter(p => activeFilters.status.includes(p.status));
if (activeFilters.format.length) products = products.filter(p => activeFilters.format.includes(p.format));
// Sort
const sort = sortOverride || currentSort;
switch (sort) {
case 'newest': products.sort((a, b) => b.id - a.id); break;
case 'price-asc': products.sort((a, b) => a.price - b.price); break;
case 'price-desc': products.sort((a, b) => b.price - a.price); break;
case 'name': products.sort((a, b) => a.name.localeCompare(b.name)); break;
case 'stock': products.sort((a, b) => b.stock - a.stock); break;
default: break;
}
return products;
}

function addToCart(id, qty = 10) {
const p = PRODUCTS.find(x => x.id === id);
if (!p || p.status === 'out-of-stock') return;
const item = cart.find(c => c.id === id);
if (item) item.qty += qty;
else cart.push({ id: p.id, name: p.name, sku: p.sku, price: parseFloat(p.price), qty, mockup: p.mockup, category: p.category });
updateCart();
toggleCart(true);
renderProducts(); // for badge updates
}
function updateCart() {
document.getElementById('cartCount').textContent = cart.reduce((s, i) => s + i.qty, 0);
const itemContainer = document.getElementById('cartItems');
if (!cart.length) {
itemContainer.innerHTML = `<div class="text-center" style="padding:40px 0;"><div style="font-size:3rem;">🛒</div><p style="margin:15px 0;">Cart is empty</p></div>`;
document.getElementById('cartSubtotal').textContent = '$0';
return;
}
itemContainer.innerHTML = cart.map((item, i) => `
<div class="cart-item">
<div class="cart-item-img"><img src="${item.mockup}" alt="${item.name}" style="width:100%; height:100%; object-fit:cover;"></div>
<div class="cart-item-info">
<div class="cart-item-title">${item.name}</div>
<div class="cart-item-sku">SKU: ${item.sku}</div>
<div class="cart-item-variants">Qty: ${item.qty} pcs</div>
<div class="cart-item-quantity">
<button class="qty-btn" onclick="changeQty(${i}, -10)">−</button>
<span class="qty-value">${item.qty}</span>
<button class="qty-btn" onclick="changeQty(${i}, 10)">+</button>
<span style="margin-left:auto; font-weight:600;">$${(item.price * item.qty).toFixed(2)}</span>
</div>
<button class="cart-item-remove" onclick="removeFromCart(${i})">Remove</button>
</div>
</div>
`).join('');
document.getElementById('cartSubtotal').textContent = '$' + cart.reduce((s, i) => s + i.price * i.qty, 0).toFixed(2);
}
function changeQty(idx, delta) {
cart[idx].qty = Math.max(10, cart[idx].qty + delta);
updateCart();
}
function removeFromCart(idx) {
cart.splice(idx, 1);
updateCart();
}

drawer.classList.add('active');

}
